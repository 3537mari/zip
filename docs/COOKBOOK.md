## Основы

### 1. Hello, World

```zip
печать_строки("Привет, мир!")
```

### 2. Арифметика

```zip
функция главная() {
    а = 10
    б = 3
    печать_число(а + б)  // 13
    печать_число(а - б)  // 7
    печать_число(а * б)  // 30
    печать_число(а / б)  // 3
}
```

### 3. Строки

```zip
функция главная() {
    имя = "Иван"
    печать_строки(имя)
}
```

### 4. Условия

```zip
функция описать_число(н) {
    если (н > 0) {
        возврат("положительное")
    } иначе если (н < 0) {
        возврат("отрицательное")
    } иначе {
        возврат("ноль")
    }
}

функция главная() {
    печать_строки(описать_число(5))
    печать_строки(описать_число(-3))
    печать_строки(описать_число(0))
}
```

### 5. Циклы

```zip
функция главная() {
    для и в 0..10 {
        печать_число(и)
    }

    счётчик = 0
    пока (счётчик < 5) {
        печать_число(счётчик)
        счётчик = счётчик + 1
    }

    для и в 0..100 {
        если (и > 5) { прервать }
        печать_число(и)
    }
}
```

## Работа с данными

### 6. Массивы

```zip
функция главная() {
    числа = СИПРА(10)
    СИПРА_добавить(числа, 10)
    СИПРА_добавить(числа, 20)
    СИПРА_добавить(числа, 30)
}
```

### 7. Итерация

```zip
функция сумма_массива(числа) {
    с = 0
    для число в СЫПРА_из_СИПРА(числа) {
        с = с + число
    }
    возврат(с)
}
```

### 8. Сортировка

```zip
функция сортировать(массив, размер) {
    для и в 0..размер {
        для j в 0..(размер - и - 1) {
            если (массив[j] > массив[j + 1]) {
                временное = массив[j]
                массив[j] = массив[j + 1]
                массив[j + 1] = временное
            }
        }
    }
}
```

### 9. Поиск

```zip
функция найти(массив, размер, значение) {
    для и в 0..размер {
        если (массив[и] == значение) {
            возврат(Нечто(и))
        }
    }
    возврат(Ничто)
}

функция бинарный_поиск(массив, размер, значение) {
    лево = 0
    право = размер - 1
    пока (лево <= право) {
        середина = (лево + право) / 2
        если (массив[середина] == значение) {
            возврат(Нечто(середина))
        } иначе если (массив[середина] < значение) {
            лево = середина + 1
        } иначе {
            право = середина - 1
        }
    }
    возврат(Ничто)
}
```

### 10. Фильтрация

```zip
функция только_чётные(массив, размер) {
    результат = СИПРА(размер)
    для и в 0..размер {
        если (массив[и] % 2 == 0) {
            СИПРА_добавить(результат, массив[и])
        }
    }
    возврат(результат)
}
```

## Функции

### 11. Рекурсия

```zip
функция факториал(н) {
    если (н < 2) { возврат(1) }
    возврат(н * факториал(н - 1))
}

функция число_фибоначчи(н) {
    если (н < 2) { возврат(н) }
    возврат(число_фибоначчи(н - 1) + число_фибоначчи(н - 2))
}
```

### 12. Специализация

```zip
глагол сложить(а: Целое, б: Целое) -> Целое {
    возврат(а + б)
}

глагол сложить(а: Строка, б: Строка) -> Строка {
    возврат(а)  // упрощённо; склейка через STD/FFI
}
```

## Работа с файлами

### 16. Чтение файла

```zip
#link("libc")
функция прочитать_файл(путь) {
    файл = fopen(путь, "r")
    если (файл == 0) {
        возврат(Ошибка("Не могу открыть файл"))
    }
    буфер = ПАМЯТЬ_ВЫДЕЛИТЬ(4096)
    прочитано = fread(буфер, 1, 4096, файл)
    fclose(файл)
    возврат(Успех(буфер))
}
```

### 17. Запись файла

```zip
#link("libc")
функция записать_файл(путь, содержимое, длина) {
    файл = fopen(путь, "w")
    если (файл == 0) {
        возврат(Ошибка("Не могу создать файл"))
    }
    fwrite(содержимое, 1, длина, файл)
    fclose(файл)
    возврат(Успех(0))
}
```

### 18. Копирование

```zip
#link("libc")
функция копировать(откуда, куда) {
    источник = fopen(откуда, "rb")
    назначение = fopen(куда, "wb")
    буфер = ПАМЯТЬ_ВЫДЕЛИТЬ(8192)
    пока (истина) {
        n = fread(буфер, 1, 8192, источник)
        если (n <= 0) { прервать }
        fwrite(буфер, 1, n, назначение)
    }
    fclose(источник)
    fclose(назначение)
}
```

### 19. Список файлов

```zip
#runtime("filesystem")
// Скелет: обход каталога через FFI (dirent / Win32)
```


### 25. Проверить доступность


## Агенты и потоки

### 36. Счётчик

```zip
агент Счётчик {
    состояние {
        всего: Целое = 0
    }
    функция увеличить() {
        всего = всего + 1
    }
    функция сбросить() {
        всего = 0
    }
}

функция главная() {
    счётчик = создать_Счётчик()
    счётчик.увеличить()
    счётчик.увеличить()
```

### 48. Нечто/Ничто

```zip
функция найти_пользователя(ид) {
    если (ид > 0) {
        возврат(Нечто(ид))
    }
    возврат(Ничто)
}

функция главная() {
    результат = найти_пользователя(5)
    если (результат — Нечто) {
        печать_число(нечто_значение(результат))
    } иначе если (результат — Ничто) {
        печать_строки("Не найдено")
    }
}
```

### 49. Успех/Ошибка

```zip
функция разделить(а, б) {
    если (б == 0) {
        возврат(Ошибка("Деление на ноль"))
    }
    возврат(Успех(а / б))
}

функция главная() {
    результат = разделить(10, 0)
    если (результат — Успех) {
        печать_число(успех_значение(результат))
    } иначе если (результат — Ошибка) {
        печать_строки(ошибка_текст(результат))
    }
}
```
