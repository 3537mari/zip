функция разобрать_блок(виды, целые, тексты, поз, нток, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, счёт) {
    если (п_есть(виды, поз, 10) == 1) { п_вперёд(поз) }
    голова = 0
    хвост = 0
    пока ((п_есть(виды, поз, 11) == 0) и (п_есть(виды, поз, 0) == 0)) {
        
        ст = разобрать_инструкцию(виды, целые, тексты, поз, нток, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, счёт)
        узел = узел_новый(у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, счёт, 20, ст, 0, 0, "", 0)
        если (голова == 0) {
            голова = узел
            хвост = узел
        } иначе {
            у_след[хвост] = узел
            хвост = узел
        }
    }
    если (п_есть(виды, поз, 11) == 1) { п_вперёд(поз) }
    возврат(узел_новый(у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, счёт, 15, голова, 0, 0, "", 0))
}

функция разобрать_функцию(виды, целые, тексты, поз, нток, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, счёт) {
    п_вперёд(поз)  // функция
    имя = п_текст(тексты, поз)
    п_вперёд(поз)
    арность = 0
    парам_голова = 0
    парам_хвост = 0
    если (п_есть(виды, поз, 8) == 1) {
        п_вперёд(поз)
        пока ((п_есть(виды, поз, 9) == 0) и (п_есть(виды, поз, 0) == 0)) {
            
            пн = п_текст(тексты, поз)
            п_вперёд(поз)
            если (п_есть(виды, поз, 7) == 1) {
                п_вперёд(поз)
                п_вперёд(поз)  // тип
            }
            узел = узел_новый(у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, счёт, 20, 0, 0, 0, пн, 0)
            если (парам_голова == 0) {
                парам_голова = узел
                парам_хвост = узел
            } иначе {
                у_след[парам_хвост] = узел
                парам_хвост = узел

            если (п_есть(виды, поз, 14) == 1) { п_вперёд(поз) }
        }
        если (п_есть(виды, поз, 9) == 1) { п_вперёд(поз) }
    }
    если (п_есть(виды, поз, 15) == 1) {
        п_вперёд(поз)
        п_вперёд(поз)

    поз = [0; 1]

    счёт[0] = 1

    хвост = 0
    пока (п_есть(виды, поз, 0) == 0) {
        п_пропуск_директив(виды, тексты, поз, нток)
        если (п_есть(виды, поз, 0) == 0) {
        видт = п_вид(виды, поз)

            п_вперёд(поз)
            если (п_есть(виды, поз, 10) == 1) {

                п_вперёд(поз)

                голова = св
                хвост = св
            } иначе {
                у_след[хвост] = св
                хвост = св
            }
        }
        }
    }
