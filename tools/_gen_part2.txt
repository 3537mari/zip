функция эмит_функцию(фн, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, стр_таб, стр_н, фн_имена, фн_н) {
    лок_имена = [0; 256]
    лок_н = [0; 1]
    лок_н[0] = 0
    ипк = [0; 1]
    ипк[0] = 0
    п = у_а[фн]
    пока (п != 0) {
        лок_добавить(лок_имена, лок_н, у_имя[п])
        п = у_след[п]
    }
    эмит_инструкцию(у_б[фн], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
    байтбуфер_дописать(код, 97)
    ипк[0] = ипк[0] + 1
    возврат(лок_н[0])
}
функция собрать_модуль(прог, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, путь_выхода) {
    стр_таб = [0; 16384]
    стр_н = [0; 1]
    стр_н[0] = 0
    фн_имена = [0; 256]
    фн_арн = [0; 256]
    фн_лок = [0; 256]
    фн_н = [0; 1]
    фн_н[0] = 0

    сп = у_а[прог]
    пока (сп != 0) {
        у = у_а[сп]
        если (у_вид[у] == 2) {
            н = фн_н[0]
            фн_имена[н] = у_имя[у]
            фн_арн[н] = у_цел[у]
            стр_добавить(стр_таб, стр_н, у_имя[у])
            фн_н[0] = н + 1
        }
        сп = у_след[сп]
    }

    // эмит всех тел в отдельные буферы, потом соберём
    // Храним длины и переэмитируем: однопроходная сборка через список байтбуферов как целые id
    коды_id = [0; 256]
    сп = у_а[прог]
    пока (сп != 0) {
        у = у_а[сп]
        если (у_вид[у] == 2) {
            индекс = фн_найти(фн_имена, фн_н, у_имя[у])
            код = байтбуфер_создать()
            лок = эмит_функцию(у, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, стр_таб, стр_н, фн_имена, фн_н)
            если (лок < 16) { лок = 16 }
            фн_лок[индекс] = лок
            коды_id[индекс] = код
        }
        сп = у_след[сп]
    }

    out = байтбуфер_создать()
    байтбуфер_дописать(out, 90)
    байтбуфер_дописать(out, 73)
    байтбуфер_дописать(out, 80)
    байтбуфер_дописать(out, 66)
    г_дописать_u16(out, 3)
    г_дописать_u16(out, 0)
    г_дописать_u32(out, фн_н[0])
    г_дописать_u32(out, 16)
    г_дописать_u32(out, стр_н[0])
    й = 0
    пока (й < стр_н[0]) {
        с = стр_таб[й]
        дл = длина_байтов(с)
        г_дописать_u32(out, дл)
        j = 0
        пока (j < дл) {
            байтбуфер_дописать(out, байт_строки(с, j))
            j = j + 1
        }
        й = й + 1
    }

    й = 0
    пока (й < фн_н[0]) {
        имя_и = 0
        k = 0
        пока (k < стр_н[0]) {
            если (сравнить_строки(стр_таб[k], фн_имена[й]) == 0) {
                имя_и = k
                k = стр_н[0]
            }
            k = k + 1
        }
        г_дописать_u32(out, имя_и)
        г_дописать_u16(out, фн_арн[й])
        г_дописать_u16(out, фн_лок[й])
        г_дописать_u32(out, 0)
        // длина и байты кода — нужен байтбуфер_слить
        код = коды_id[й]
        дл = байтбуфер_длина(код)
        г_дописать_u32(out, дл)
        байтбуфер_слить(out, код)
        й = й + 1
    }

    байтбуфер_сохранить(out, путь_выхода)
    возврат(байтбуфер_длина(out))
}

функция компилировать_файл(вход, выход) {
    исходник = файл_прочитать(вход)
    виды = [0; 16384]
    целые = [0; 16384]
    тексты = [0; 16384]
    нток = лексер_разобрать(исходник, виды, целые, тексты)
    напечатать_строку("Токенов:")
    напечатать_число_строкой(нток)

    у_вид = [0; 16384]
    у_а = [0; 16384]
    у_б = [0; 16384]
    у_в = [0; 16384]
    у_имя = [0; 16384]
    у_цел = [0; 16384]
    у_след = [0; 16384]
    счёт = [0; 1]
    счёт[0] = 1
    прог = парсер_разобрать(виды, целые, тексты, нток, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, счёт)
    напечатать_строку("AST узлов:")
    напечатать_число_строкой(счёт[0])

    дл = собрать_модуль(прог, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, выход)
    напечатать_строку("Байт-код:")
    напечатать_число_строкой(дл)
    возврат(0)
}

функция главная() {
    если (количество_аргументов() < 3) {
        напечатать_ошибку("Использование: selfc <вход.zip> <выход.zipb>")
        возврат(1)
    }
    вход = получить_аргумент(1)
    выход = получить_аргумент(2)
    напечатать_строку("Компиляция ZIP → ZIPB (чистый ZIP):")
    напечатать_строку(вход)
    код = компилировать_файл(вход, выход)
    если (код != 0) {
        напечатать_ошибку("Ошибка компиляции")
        возврат(код)
    }
    напечатать_строку("OK:")
    напечатать_строку(выход)
    возврат(0)
}
