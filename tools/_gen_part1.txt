// === Генератор ZIPB ===
функция г_дописать_u16(б, з) {
    байтбуфер_дописать(б, з - (з / 256) * 256)
    байтбуфер_дописать(б, з / 256)
}
функция г_дописать_u32(б, з) {
    г_дописать_u16(б, з - (з / 65536) * 65536)
    г_дописать_u16(б, з / 65536)
}
функция г_дописать_i64(б, з) {
    х = з
    й = 0
    пока (й < 8) {
        байт = х - (х / 256) * 256
        если (байт < 0) { байт = байт + 256 }
        байтбуфер_дописать(б, байт)
        х = х / 256
        й = й + 1
    }
}


функция эмит_оп(код, ипк, оп) {
    байтбуфер_дописать(код, оп)
    ипк[0] = ипк[0] + 1
}
функция эмит_оп_u32(код, ипк, оп, з) {
    байтбуфер_дописать(код, оп)
    г_дописать_u32(код, з)
    ипк[0] = ипк[0] + 1
}
функция эмит_оп_i64(код, ипк, оп, з) {
    байтбуфер_дописать(код, оп)
    г_дописать_i64(код, з)
    ипк[0] = ипк[0] + 1
}
функция эмит_прыжок(код, ипк, оп) {
    байтбуфер_дописать(код, оп)
    поз = байтбуфер_длина(код)
    г_дописать_u32(код, 0)
    ипк[0] = ипк[0] + 1
    возврат(поз)
}

функция стр_добавить(стр_таб, стр_н, s) {
    й = 0
    н = стр_н[0]
    пока (й < н) {
        если (сравнить_строки(стр_таб[й], s) == 0) { возврат(й) }
        й = й + 1
    }
    стр_таб[н] = s
    стр_н[0] = н + 1
    возврат(н)
}

функция фн_найти(фн_имена, фн_н, имя) {
    й = 0
    пока (й < фн_н[0]) {
        если (сравнить_строки(фн_имена[й], имя) == 0) { возврат(й) }
        й = й + 1
    }
    возврат(0 - 1)
}

функция лок_найти(лок_имена, лок_н, имя) {
    й = 0
    пока (й < лок_н[0]) {
        если (сравнить_строки(лок_имена[й], имя) == 0) { возврат(й) }
        й = й + 1
    }
    возврат(0 - 1)
}

функция лок_добавить(лок_имена, лок_н, имя) {
    найд = лок_найти(лок_имена, лок_н, имя)
    если (найд >= 0) { возврат(найд) }
    н = лок_н[0]
    лок_имена[н] = имя
    лок_н[0] = н + 1
    возврат(н)
}

функция эмит_выражение(у, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н) {
    если (у == 0) {
        байтбуфер_дописать(код, 16)
        г_дописать_i64(код, 0)
        ипк[0] = ипк[0] + 1
        возврат(0)
    }
    вид = у_вид[у]
    если (вид == 5) {
        байтбуфер_дописать(код, 16)
        г_дописать_i64(код, у_цел[у])
        ипк[0] = ипк[0] + 1
        возврат(0)
    }
    если (вид == 19) {
        байтбуфер_дописать(код, 16)
        г_дописать_i64(код, у_цел[у])
        ипк[0] = ипк[0] + 1
        возврат(0)
    }
    если (вид == 6) {
        байтбуфер_дописать(код, 18)
        ипк[0] = ипк[0] + 1
        инд = стр_добавить(стр_таб, стр_н, у_имя[у])
        г_дописать_u32(код, инд)
        возврат(0)
    }
    если (вид == 7) {
        слот = лок_найти(лок_имена, лок_н, у_имя[у])
        если (слот < 0) { слот = лок_добавить(лок_имена, лок_н, у_имя[у]) }
        байтбуфер_дописать(код, 64)
        г_дописать_u32(код, слот)
        ипк[0] = ипк[0] + 1
        возврат(0)
    }
    если (вид == 8) {
        эмит_выражение(у_а[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        эмит_выражение(у_б[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        оп = у_имя[у]
        если (сравнить_строки(оп, "+") == 0) { байтбуфер_дописать(код, 32) }
        иначе если (сравнить_строки(оп, "-") == 0) { байтбуфер_дописать(код, 33) }
        иначе если (сравнить_строки(оп, "*") == 0) { байтбуфер_дописать(код, 34) }
        иначе если (сравнить_строки(оп, "/") == 0) { байтбуфер_дописать(код, 35) }
        иначе если (сравнить_строки(оп, "==") == 0) { байтбуфер_дописать(код, 48) }
        иначе если (сравнить_строки(оп, "!=") == 0) { байтбуфер_дописать(код, 49) }
        иначе если (сравнить_строки(оп, "<") == 0) { байтбуфер_дописать(код, 50) }
        иначе если (сравнить_строки(оп, "<=") == 0) { байтбуфер_дописать(код, 51) }
        иначе если (сравнить_строки(оп, ">") == 0) { байтбуфер_дописать(код, 52) }
        иначе если (сравнить_строки(оп, ">=") == 0) { байтбуфер_дописать(код, 53) }
        иначе если (сравнить_строки(оп, "и") == 0) { байтбуфер_дописать(код, 34) }
        иначе если (сравнить_строки(оп, "или") == 0) { байтбуфер_дописать(код, 32) }
        ипк[0] = ипк[0] + 1
        возврат(0)
    }
    если (вид == 9) {
        эмит_выражение(у_а[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        если (сравнить_строки(у_имя[у], "-") == 0) {
            байтбуфер_дописать(код, 16)
            г_дописать_i64(код, 0)
            ипк[0] = ипк[0] + 1
            байтбуфер_дописать(код, 33)
            ипк[0] = ипк[0] + 1
        }
        возврат(0)
    }
    если (вид == 4) {
        арг = у_а[у]
        пока (арг != 0) {
            эмит_выражение(у_а[арг], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            арг = у_след[арг]
        }
        индекс = фн_найти(фн_имена, фн_н, у_имя[у])
        если (индекс >= 0) {
            байтбуфер_дописать(код, 96)
            г_дописать_u32(код, индекс)
            ипк[0] = ипк[0] + 1
        } иначе {
            байтбуфер_дописать(код, 98)
            ипк[0] = ипк[0] + 1
            либ = стр_добавить(стр_таб, стр_н, "builtin")
            ф = стр_добавить(стр_таб, стр_н, у_имя[у])
            г_дописать_u32(код, либ)
            г_дописать_u32(код, ф)
        }
        возврат(0)
    }
    если (вид == 16) {
        если (у_в[у] == 1) {
            // [val; N] — размер только в операнде NEW_ARRAY, не на стеке
            если (у_вид[у_б[у]] == 5) {
                байтбуфер_дописать(код, 151)
                г_дописать_u32(код, у_цел[у_б[у]])
                ипк[0] = ипк[0] + 1
            } иначе {
                байтбуфер_дописать(код, 151)
                г_дописать_u32(код, 16)
                ипк[0] = ипк[0] + 1
            }
        } иначе {
            байтбуфер_дописать(код, 151)
            г_дописать_u32(код, 8)
            ипк[0] = ипк[0] + 1
        }
        возврат(0)
    }
    если (вид == 17) {
        эмит_выражение(у_а[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        эмит_выражение(у_б[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        байтбуфер_дописать(код, 152)
        ипк[0] = ипк[0] + 1
        возврат(0)
    }
    байтбуфер_дописать(код, 16)
    г_дописать_i64(код, 0)
    ипк[0] = ипк[0] + 1
    возврат(0)
}

функция эмит_инструкцию(у, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н) {
    если (у == 0) { возврат(0) }
    вид = у_вид[у]
    если (вид == 14) {
        если (у_а[у] != 0) {
            эмит_выражение(у_а[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        } иначе {
            байтбуфер_дописать(код, 16)
            г_дописать_i64(код, 0)
            ипк[0] = ипк[0] + 1
        }
        байтбуфер_дописать(код, 97)
        ипк[0] = ипк[0] + 1
        возврат(0)
    }
    если (вид == 10) {
        лев = у_а[у]
        если (у_вид[лев] == 7) {
            эмит_выражение(у_б[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            слот = лок_добавить(лок_имена, лок_н, у_имя[лев])
            байтбуфер_дописать(код, 65)
            г_дописать_u32(код, слот)
            ипк[0] = ипк[0] + 1
        } иначе если (у_вид[лев] == 17) {
            // стек: массив, значение, индекс
            эмит_выражение(у_а[лев], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            эмит_выражение(у_б[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            эмит_выражение(у_б[лев], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            байтбуфер_дописать(код, 153)
            ипк[0] = ипк[0] + 1
        } иначе {
            эмит_выражение(у_б[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            байтбуфер_дописать(код, 65)
            г_дописать_u32(код, 0)
            ипк[0] = ипк[0] + 1
        }
        возврат(0)
    }
    если (вид == 11) {
        эмит_выражение(у_а[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        поз_иначе = эмит_прыжок(код, ипк, 82)
        эмит_инструкцию(у_б[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        если (у_в[у] != 0) {
            поз_конца = эмит_прыжок(код, ипк, 80)
            байтбуфер_патч_u32(код, поз_иначе, ипк[0])
            эмит_инструкцию(у_в[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            байтбуфер_патч_u32(код, поз_конца, ипк[0])
        } иначе {
            байтбуфер_патч_u32(код, поз_иначе, ипк[0])
        }
        возврат(0)
    }
    если (вид == 12) {
        нач = ипк[0]
        эмит_выражение(у_а[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        поз_кон = эмит_прыжок(код, ипк, 82)
        эмит_инструкцию(у_б[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
        байтбуфер_дописать(код, 80)
        г_дописать_u32(код, нач)
        ипк[0] = ипк[0] + 1
        байтбуфер_патч_u32(код, поз_кон, ипк[0])
        возврат(0)
    }
    если (вид == 15) {
        сп = у_а[у]
        пока (сп != 0) {
            эмит_инструкцию(у_а[сп], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            сп = у_след[сп]
        }
        возврат(0)
    }
    если (вид == 13) {
        // для й в 0..N { body } → упрощённый while
        // кол — диапазон
        кол = у_а[у]
        если (у_вид[кол] == 18) {
            слот = лок_добавить(лок_имена, лок_н, у_имя[у])
            эмит_выражение(у_а[кол], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            байтбуфер_дописать(код, 65)
            г_дописать_u32(код, слот)
            ипк[0] = ипк[0] + 1
            нач = ипк[0]
            байтбуфер_дописать(код, 64)
            г_дописать_u32(код, слот)
            ипк[0] = ипк[0] + 1
            эмит_выражение(у_б[кол], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            байтбуфер_дописать(код, 50)
            ипк[0] = ипк[0] + 1
            поз_кон = эмит_прыжок(код, ипк, 82)
            эмит_инструкцию(у_б[у], у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
            байтбуфер_дописать(код, 64)
            г_дописать_u32(код, слот)
            ипк[0] = ипк[0] + 1
            байтбуфер_дописать(код, 16)
            г_дописать_i64(код, 1)
            ипк[0] = ипк[0] + 1
            байтбуфер_дописать(код, 32)
            ипк[0] = ипк[0] + 1
            байтбуфер_дописать(код, 65)
            г_дописать_u32(код, слот)
            ипк[0] = ипк[0] + 1
            байтбуфер_дописать(код, 80)
            г_дописать_u32(код, нач)
            ипк[0] = ипк[0] + 1
            байтбуфер_патч_u32(код, поз_кон, ипк[0])
        }
        возврат(0)
    }
    // выражение как инструкция — убрать результат со стека
    эмит_выражение(у, у_вид, у_а, у_б, у_в, у_имя, у_цел, у_след, код, ипк, стр_таб, стр_н, фн_имена, фн_н, лок_имена, лок_н)
    байтбуфер_дописать(код, 1)
    ипк[0] = ипк[0] + 1
    возврат(0)
}
